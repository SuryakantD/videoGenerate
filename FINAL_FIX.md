### Image Generation:

// Pollinations API - Works perfectly!
const imageUrl = `https://image.pollinations.ai/prompt/${encodeURIComponent(prompt)}?model=flux&width=1024&height=576&seed=${seed}&nologo=true`;

### Video Generation:
```typescript
// Create canvas
const canvas = document.createElement('canvas');
canvas.width = 1024;
canvas.height = 576;
const ctx = canvas.getContext('2d');

// Capture stream at 30 FPS
const stream = canvas.captureStream(30);

// Setup MediaRecorder
const recorder = new MediaRecorder(stream, {
  mimeType: 'video/webm;codecs=vp9',
  videoBitsPerSecond: 5000000, // 5 Mbps
});

// Animate images with Ken Burns effect
const animate = () => {
  // Draw current image with zoom effect
  const scale = 1 + progress * 0.1; // 1.0x to 1.1x
  ctx.drawImage(img, offsetX, offsetY, width * scale, height * scale);
  
  // Add fade transitions
  if (transitionInProgress) {
    ctx.fillStyle = `rgba(0, 0, 0, ${fadeAlpha})`;
    ctx.fillRect(0, 0, width, height);
  }
  
  requestAnimationFrame(animate);
};

// Record and save
recorder.start();
animate();
// ... after duration ...
recorder.stop();
const blob = new Blob(chunks, { type: 'video/webm' });
const videoUrl = URL.createObjectURL(blob);
```
